### 4.1 证书约束

```sql
DECLARE @ck sysname, @sql nvarchar(max);
SELECT TOP 1 @ck = name FROM sys.check_constraints
 WHERE parent_object_id = OBJECT_ID('dbo.service_item')
   AND definition LIKE '%certificate_type%';
IF @ck IS NOT NULL
BEGIN
    SET @sql = N'ALTER TABLE dbo.service_item DROP CONSTRAINT ' + QUOTENAME(@ck);
    EXEC sp_executesql @sql;
END;
ALTER TABLE dbo.service_item ADD CONSTRAINT CK_service_item_certificate CHECK (
    (certificate_type IS NULL AND certificate_level IS NULL)
    OR (certificate_type IS NOT NULL AND certificate_level IS NOT NULL AND certificate_level BETWEEN 1 AND 3)
);
```

CREATE INDEX IX_assignment_shift ON dbo.assignment(shift_id, status) INCLUDE(is_current);
